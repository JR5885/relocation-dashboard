"""
GM-Robot :: compliance_check.py
Cross-checks HR + Lawyer feeds against statutory visa/labor deadlines.
Escalates breaches to Slack and syncs live status to Notion.
"""
 
import pandas as pd
from datetime import date
from notion_client import Client as NotionClient
from slack_sdk import WebClient as SlackClient
 
STAGE_SLA_DAYS = {
    "Entity Card Setup": 10,
    "Veremark Pending": 5,
    "Veremark Completed": 3,
    "EP / RC Submission": 7,
    "EP / RC Approval": 10,
}
 
notion = NotionClient(auth=SECRETS["NOTION_TOKEN"])
slack = SlackClient(token=SECRETS["SLACK_BOT_TOKEN"])
 
 
def load_source_feeds():
    """Merge HR roster, lawyer tracker, and Veremark export into one frame."""
    hr = pd.read_csv("feeds/hr_roster.csv")
    legal = pd.read_csv("feeds/lawyer_tracker.csv")
    veremark = pd.read_csv("feeds/veremark_export.csv")
    merged = hr.merge(legal, on="employee_id", how="left")
    merged = merged.merge(veremark, on="employee_id", how="left")
    return merged
 
 
def flag_breaches(df: pd.DataFrame) -> pd.DataFrame:
    """Mark any case that has exceeded its stage SLA."""
    today = date.today()
    df["days_in_stage"] = (today - pd.to_datetime(df["stage_entered_on"]).dt.date).dt.days
    df["sla_days"] = df["stage"].map(STAGE_SLA_DAYS)
    df["breach"] = df["days_in_stage"] > df["sla_days"]
    return df
 
 
def escalate(breaches: pd.DataFrame) -> None:
    for _, row in breaches.iterrows():
        message = (
            f":rotating_light: *{row['region']}* — {row['employee_name']} "
            f"stuck at *{row['stage']}* for {row['days_in_stage']}d "
            f"(SLA {row['sla_days']}d). Owner: {row['vendor']}"
        )
        slack.chat_postMessage(channel="#relocation-blitz-alerts", text=message)
 
 
def sync_to_notion(df: pd.DataFrame) -> None:
    for _, row in df.iterrows():
        notion.pages.update(
            page_id=row["notion_page_id"],
            properties={
                "Stage": {"select": {"name": row["stage"]}},
                "Compliance": {"select": {"name": "Breach" if row["breach"] else "OK"}},
            },
        )
 
 
def run() -> None:
    df = flag_breaches(load_source_feeds())
    escalate(df[df["breach"]])
    sync_to_notion(df)
    print(f"Checked {len(df)} cases — {df['breach'].sum()} breach(es) escalated.")
 
 
if __name__ == "__main__":
    run()
